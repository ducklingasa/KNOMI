cmake_minimum_required(VERSION 3.20)
project(knomi_screen_reference C)

set(CMAKE_C_STANDARD 99)
get_filename_component(KNOMI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(REFERENCE_SOURCE_ROOT "${KNOMI_ROOT}" CACHE PATH "Root containing the legacy src/ui sources")
if(NOT EXISTS "${REFERENCE_SOURCE_ROOT}/src/ui/ui.h")
    file(READ "${KNOMI_ROOT}/docs/screens/original/legacy-source.json" SOURCE_MANIFEST)
    string(JSON EXPECTED_ARCHIVE_HASH GET "${SOURCE_MANIFEST}" sha256)
    file(SHA256 "${KNOMI_ROOT}/docs/screens/original/legacy-source.zip" ACTUAL_ARCHIVE_HASH)
    if(NOT EXPECTED_ARCHIVE_HASH STREQUAL ACTUAL_ARCHIVE_HASH)
        message(FATAL_ERROR "Legacy visual source archive hash mismatch")
    endif()
    set(REFERENCE_SOURCE_ROOT "${CMAKE_CURRENT_BINARY_DIR}/legacy_source")
    file(ARCHIVE_EXTRACT INPUT "${KNOMI_ROOT}/docs/screens/original/legacy-source.zip"
        DESTINATION "${REFERENCE_SOURCE_ROOT}")
endif()
set(DEFAULT_LVGL_SOURCE "${KNOMI_ROOT}/managed_components/lvgl__lvgl")
if(NOT EXISTS "${DEFAULT_LVGL_SOURCE}/lvgl.h")
    set(DEFAULT_LVGL_SOURCE "${KNOMI_ROOT}/spikes/display/managed_components/lvgl__lvgl")
endif()
set(LVGL_SOURCE_DIR "${DEFAULT_LVGL_SOURCE}" CACHE PATH "LVGL 8.4.0 source directory")
if(NOT EXISTS "${LVGL_SOURCE_DIR}/lvgl.h")
    message(FATAL_ERROR "Set LVGL_SOURCE_DIR to an existing LVGL 8.4.0 source checkout")
endif()

add_compile_definitions(LV_CONF_INCLUDE_SIMPLE LV_KCONFIG_IGNORE)
include_directories("${CMAKE_CURRENT_LIST_DIR}" "${LVGL_SOURCE_DIR}")
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_SOURCE_DIR}/src/*.c")
add_library(reference_lvgl OBJECT ${LVGL_SOURCES})

file(STRINGS "${REFERENCE_SOURCE_ROOT}/src/ui/ui.h" UI_GLOBALS REGEX "^extern lv_obj_t \\*")
file(STRINGS "${REFERENCE_SOURCE_ROOT}/src/ui/ui.h" UI_EVENTS REGEX "^void ui_event_.*\\);")
set(UI_ADAPTER "#include \"ui.h\"\n")
foreach(DECLARATION IN LISTS UI_GLOBALS)
    string(REGEX REPLACE ";[ \t]*$" "" DECLARATION "${DECLARATION}")
    string(REPLACE "extern " "" DEFINITION "${DECLARATION}")
    string(APPEND UI_ADAPTER "${DEFINITION};\n")
endforeach()
foreach(DECLARATION IN LISTS UI_EVENTS)
    string(REGEX REPLACE ";[ \t]*$" "" DECLARATION "${DECLARATION}")
    string(REGEX REPLACE "\\)$" ") { (void)e; }" DEFINITION "${DECLARATION}")
    string(APPEND UI_ADAPTER "${DEFINITION}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ui_adapter.c" "${UI_ADAPTER}")

set(REFERENCE_SCREEN_NAMES HeatingNozzle HeatingBed Printing Temp MainGif Welcome
    WIFIConnecting WIFIDisconnect QRCode Info Popup Dialog)
set(REFERENCE_SCREENS "")
foreach(NAME IN LISTS REFERENCE_SCREEN_NAMES)
    list(APPEND REFERENCE_SCREENS "${REFERENCE_SOURCE_ROOT}/src/ui/screens/ui_Screen${NAME}.c")
endforeach()
file(GLOB REFERENCE_IMAGES CONFIGURE_DEPENDS "${REFERENCE_SOURCE_ROOT}/src/ui/images/*.c")
file(GLOB REFERENCE_FONTS CONFIGURE_DEPENDS "${REFERENCE_SOURCE_ROOT}/src/ui/fonts/*.c")
add_executable(screen_reference main.c "${CMAKE_CURRENT_BINARY_DIR}/ui_adapter.c"
    ${REFERENCE_SCREENS} ${REFERENCE_IMAGES} ${REFERENCE_FONTS}
    "${REFERENCE_SOURCE_ROOT}/src/gif/gif_wifi.c"
    "${REFERENCE_SOURCE_ROOT}/src/gif/gif_voron.c")
target_include_directories(screen_reference PRIVATE "${REFERENCE_SOURCE_ROOT}/src/ui")
target_link_libraries(screen_reference PRIVATE reference_lvgl)
if(NOT WIN32)
    target_link_libraries(screen_reference PRIVATE m)
endif()
